/* ========================================================= WINDOWS HELLO — OVERLAY LOADING MODULE Dán vào CUỐI file app.js ========================================================= */ (function initWindowsHelloOverlay() { 'use strict'; // ---------- 1. HÀM ĐIỀU KHIỂN OVERLAY ---------- function showHelloLoading(text, subtext) { const overlay = document.getElementById('helloLoadingOverlay'); if (!overlay) return; if (text) overlay.querySelector('.hello-loading-text').textContent = text; if (subtext !== undefined) { overlay.querySelector('.hello-loading-subtext').textContent = subtext; } overlay.classList.remove('hidden'); document.body.classList.add('hello-locked'); } function hideHelloLoading() { const overlay = document.getElementById('helloLoadingOverlay'); if (!overlay) return; overlay.classList.add('hidden'); document.body.classList.remove('hello-locked'); } // Cho phép gọi từ nơi khác nếu cần window.showHelloLoading = showHelloLoading; window.hideHelloLoading = hideHelloLoading; // ---------- 2. KIỂM TRA HỖ TRỢ ---------- function isWebAuthnSupported() { return !!(window.PublicKeyCredential && navigator.credentials); } // ---------- 3. HÀM GỌI XÁC THỰC ---------- async function performWindowsHelloAuth() { // Bước 1: Kiểm tra hỗ trợ if (!isWebAuthnSupported()) { throw Object.assign(new Error('Trình duyệt không hỗ trợ Windows Hello / WebAuthn.'), { name: 'NotSupportedError' }); } // Bước 2: Hiện overlay ngay showHelloLoading('Đang kết nối với Windows Hello...', 'Vui lòng không đóng cửa sổ này'); // Bước 3: Lấy challenge từ server (nếu có API) let challenge; try { const chalRes = await fetch('/api/hello/challenge', { method: 'POST', credentials: 'include', headers: { 'Content-Type': 'application/json' } }); if (chalRes.ok) { const j = await chalRes.json(); // challenge thường là base64url challenge = base64UrlToUint8Array(j.challenge); } else { throw new Error('no-api'); } } catch (e) { // Fallback: dùng challenge ngẫu nhiên (chỉ để demo UI) console.warn('[Windows Hello] Không có API challenge, dùng challenge tạm thời.'); challenge = new Uint8Array(32); crypto.getRandomValues(challenge); } // Bước 4: Gọi Windows Hello (native dialog của HĐH) showHelloLoading('Đang chờ xác thực Windows Hello...', 'Hãy quét vân tay hoặc khuôn mặt'); const credential = await navigator.credentials.get({ publicKey: { challenge, timeout: 60000, userVerification: 'required', // rpId mặc định là domain hiện tại } }); // Bước 5: Gửi credential về server để xác thực showHelloLoading('Đang xác thực với server...', 'Vui lòng đợi trong giây lát'); const payload = { id: credential.id, rawId: uint8ArrayToBase64(credential.rawId), type: credential.type, response: { clientDataJSON: uint8ArrayToBase64(credential.response.clientDataJSON), authenticatorData: uint8ArrayToBase64(credential.response.authenticatorData), signature: uint8ArrayToBase64(credential.response.signature), userHandle: credential.response.userHandle ? uint8ArrayToBase64(credential.response.userHandle) : null } }; let serverOk = false; try { const res = await fetch('/api/hello/verify', { method: 'POST', credentials: 'include', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (res.ok) { serverOk = true; } else { const data = await res.json().catch(() => ({})); throw new Error(data.error || 'Server từ chối xác thực.'); } } catch (err) { // Nếu chưa có API server, vẫn cho phép thành công cục bộ (dev mode) if (err.message && err.message.includes('Server từ chối')) throw err; console.warn('[Windows Hello] Chưa có API verify, bỏ qua bước server (dev mode).'); serverOk = true; } // Bước 6: Thành công showHelloLoading('Xác thực thành công!', 'Đang mở khóa ứng dụng...'); return { credential, serverOk }; } // ---------- 4. HELPER BASE64 ---------- function uint8ArrayToBase64(arr) { let s = ''; const bytes = new Uint8Array(arr); for (let i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]); return btoa(s); } function base64UrlToUint8Array(b64url) { const b64 = b64url.replace(/-/g, '+').replace(/_/g, '/'); const pad = b64.length % 4 === 0 ? '' : '='.repeat(4 - (b64.length % 4)); const raw = atob(b64 + pad); const out = new Uint8Array(raw.length); for (let i = 0; i < raw.length; i++) out[i] = raw.charCodeAt(i); return out; } // ---------- 5. GẮN SỰ KIỆN CHO NÚT ---------- function bindButton() { const btn = document.getElementById('btnBiometricUnlock'); if (!btn) return; // Tránh gắn trùng nếu app.js bị load 2 lần if (btn.dataset.helloBound === '1') return; btn.dataset.helloBound = '1'; btn.addEventListener('click', async (e) => { e.preventDefault(); e.stopPropagation(); // Vô hiệu hóa nút để tránh double click btn.disabled = true; btn.classList.add('opacity-60', 'cursor-not-allowed'); try { const result = await performWindowsHelloAuth(); // Gọi callback mở khóa nếu app.js của bạn có hàm này // (thường là hàm unlock app / ẩn lockScreen) if (typeof window.onBiometricSuccess === 'function') { await window.onBiometricSuccess(result); } else if (typeof window.unlockApp === 'function') { await window.unlockApp(); } else { // Fallback: tự ẩn lockScreen const lock = document.getElementById('lockScreen'); if (lock) { lock.classList.add('hidden-lock'); } } // Ẩn overlay sau khi xong setTimeout(hideHelloLoading, 400); } catch (err) { hideHelloLoading(); handleHelloError(err); } finally { btn.disabled = false; btn.classList.remove('opacity-60', 'cursor-not-allowed'); } }); } // ---------- 6. XỬ LÝ LỖI ---------- function handleHelloError(err) { console.error('[Windows Hello]', err); const name = err && err.name; let msg = 'Không thể xác thực Windows Hello.'; if (name === 'NotAllowedError') { msg = 'Bạn đã hủy xác thực Windows Hello.'; } else if (name === 'SecurityError') { msg = 'Windows Hello yêu cầu HTTPS hoặc localhost.'; } else if (name === 'NotSupportedError') { msg = 'Trình duyệt không hỗ trợ Windows Hello.'; } else if (name === 'AbortError') { msg = 'Quá thời gian chờ Windows Hello (60s).'; } else if (err && err.message) { msg = err.message; } // Ưu tiên dùng toast của app nếu có if (typeof window.showToast === 'function') { window.showToast(msg, 'error'); } else { alert(msg); } } // ---------- 7. KHỞI TẠO ---------- if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', bindButton); } else { bindButton(); } // Nếu sau này nút được render động, dùng MutationObserver const observer = new MutationObserver(() => { const btn = document.getElementById('btnBiometricUnlock'); if (btn && btn.dataset.helloBound !== '1') bindButton(); }); observer.observe(document.body, { childList: true, subtree: true }); console.log('[Windows Hello] Overlay loading đã sẵn sàng.'); })();
